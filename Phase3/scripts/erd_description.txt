ERD DESCRIPTION

ENTITIES
============================================================

ENTITY: STATUS
----------------------------------------
PRIMARY KEYS:
  - status_id

ATTRIBUTES:
  - status_id (INT)
  - status_label (VARCHAR(50))


ENTITY: REQUESTCATEGORY
----------------------------------------
PRIMARY KEYS:
  - Category_id

ATTRIBUTES:
  - Category_id (INT)
  - Category_name (VARCHAR(100))
  - description (TEXT)
  - required_skills (VARCHAR(255))


ENTITY: LOCATION
----------------------------------------
PRIMARY KEYS:
  - latitude
  - longitude

ATTRIBUTES:
  - latitude (NUMERIC(9,6))
  - longitude (NUMERIC(9,6))
  - city (VARCHAR(100))
  - street (VARCHAR(100))
  - house_number (INT)


ENTITY: FAMILY
----------------------------------------
PRIMARY KEYS:
  - ContactPerson_id

ATTRIBUTES:
  - ContactPerson_id (INT)
  - ContactPerson_name (VARCHAR(255))
  - phone_number (VARCHAR(20))
  - -- (Constraint:)
  - number_of_members (INT)
  - special_features (TEXT)


ENTITY: VOLUNTEER
----------------------------------------
PRIMARY KEYS:
  - volunteer_id

ATTRIBUTES:
  - volunteer_id (INT)
  - first_name (VARCHAR(100))
  - last_name (VARCHAR(100))
  - phone_number (VARCHAR(20))
  - has_equipment (BOOLEAN)
  - availability_status (VARCHAR(50))
  - counter (INT)
  - skill_type (VARCHAR(100))
  - latitude (NUMERIC(9,6))
  - longitude (NUMERIC(9,6))

FOREIGN KEYS:
  - latitude, longitude REFERENCES LOCATION(latitude, longitude)


ENTITY: REQUEST
----------------------------------------
PRIMARY KEYS:
  - Request_id

ATTRIBUTES:
  - Request_id (INT)
  - date (DATE)
  - image (VARCHAR(255),)
  - incident_description (TEXT)
  - -- (Constraint:)
  - prioriry_level (INT)
  - ContactPerson_id (INT)
  - Category_id (INT)
  - status_id (INT)
  - --treatment_id (INT)
  - latitude (NUMERIC(9,6))
  - longitude (NUMERIC(9,6))

FOREIGN KEYS:
  - ContactPerson_id REFERENCES FAMILY(ContactPerson_id)
  - Category_id REFERENCES REQUESTCATEGORY(Category_id)
  - status_id REFERENCES STATUS(status_id)
  - treatment_id REFERENCES TREATMENT(treatment_id)
  - latitude, longitude REFERENCES LOCATION(latitude, longitude)


ENTITY: DELIVERY
----------------------------------------
PRIMARY KEYS:
  - delivery_id

ATTRIBUTES:
  - delivery_id (INT)
  - date (DATE)
  - status (VARCHAR(50))
  - item_type (VARCHAR(100))
  - -- (Constraint:)
  - quantity (INT)
  - Request_id (INT)

FOREIGN KEYS:
  - Request_id REFERENCES REQUEST(Request_id)


ENTITY: TREATMENT
----------------------------------------
PRIMARY KEYS:
  - treatment_id

ATTRIBUTES:
  - treatment_id (INT)
  - date (DATE)
  - start_time (TIME)
  - completion_time (TIME)
  - feedback_notes (TEXT)
  - photo_after (VARCHAR(255))
  - delivery_id (INT)
  - volunteer_id (INT)
  - request_id (INT)

FOREIGN KEYS:
  - delivery_id REFERENCES DELIVERY(delivery_id)
  - volunteer_id REFERENCES VOLUNTEER(volunteer_id)
  - request_id REFERENCES REQUEST(Request_id)
